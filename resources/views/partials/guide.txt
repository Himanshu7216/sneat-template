namespace Database\Seeders;

use Illuminate\Database\Seeder;

use Spatie\Permission\Models\Permission;

class RoleAndPermissionSeeder extends Seeder
{
    public function run(): void
    {
        // Reset cached roles and permissions
        app()[\Spatie\Permission\PermissionRegistrar::class]->forgetCachedPermissions();

        // 1. Create Permissions
        $editArticles = Permission::create(['name' => 'edit articles']);
        $deleteArticles = Permission::create(['name' => 'delete articles']);
        $publishArticles = Permission::create(['name' => 'publish articles']);

        // 2. Create Roles and Assign Permissions
        $adminRole = Role::create(['name' => 'admin']);
        $adminRole->givePermissionTo([$editArticles, $deleteArticles, $publishArticles]);

        $editorRole = Role::create(['name' => 'editor']);
        $editorRole->givePermissionTo($editArticles);

        // 3. Create Demo Users and Assign Roles
        $adminUser = User::factory()->create([
            'name' => 'Admin User',
            'email' => 'admin@example.com',
            'password' => bcrypt('password')
        ]);
        $adminUser->assignRole($adminRole);

        $editorUser = User::factory()->create([
            'name' => 'Editor User',
            'email' => 'editor@example.com',
            'password' => bcrypt('password')
        ]);
        $editorUser->assignRole($editorRole);
    }
}

<?php

namespace App\Models;

// use Illuminate\Contracts\Auth\MustVerifyEmail;
use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Foundation\Auth\User as Authenticatable;
use Illuminate\Notifications\Notifiable;
use Laravel\Sanctum\HasApiTokens;

class User extends Authenticatable
{
    use HasApiTokens, HasFactory, Notifiable;

    /**
     * The attributes that are mass assignable.
     *
     * @var array<int, string>
     */
    protected $fillable = [
        'name',
        'email',
        'password',
        'offline_registration',
        'permissions',
    ];

    /**
     * The attributes that should be hidden for serialization.
     *
     * @var array<int, string>
     */
    protected $hidden = [
        'password',
        'remember_token',
    ];

    /**
     * The attributes that should be cast.
     *
     * @var array<string, string>
     */
    protected $casts = [
        'email_verified_at' => 'datetime',
        'password' => 'hashed',
        'permissions' => 'array',
    ];

    public function hasPermission($permission)
    {
        if (!$this->permissions)
            return false;
        return in_array($permission, $this->permissions);
    }

    /**
     * Get the first route the user has access to.
     */
    public function getFirstRedirectRoute()
    {
        // Define a priority order for redirects
        $routes = [
            'dashboard' => 'admin.dashboard_new',
            'registrations' => 'admin.registration',
            'offline_registrations' => 'offline_registration.index',
            'sadhak' => 'admin.sadhak',
            'seva' => 'seva.index',
            'sevak' => 'sevak.index',
            'attend_sevak' => 'attend.sevak.index',
            'events' => 'event.show',
            'locations' => 'location.index',
            'sub_locations' => 'sub_location.show',
            'bus_details' => 'bus.details.index',
            'feedback' => 'front.feedback.index',
            'application' => 'front.application.index',
        ];

        foreach ($routes as $perm => $route) {
            if ($this->hasPermission($perm)) {
                return $route;
            }
        }

        // Special case for User Management (ID 6)
        if ($this->id == 6) {
            return 'users.index';
        }

        return 'admin.login'; // Fallback
    }
}

This approach is a custom JSON-based permission system. It does not use external packages like Spatie. Instead, it saves permissions directly into a text/JSON column inside your default users table.Here is exactly how it works, how to set it up in your database, and how to use it.How It Works Under the HoodDatabase Storage: Permissions are stored as a JSON array string in your database column (e.g., ["dashboard", "registrations", "seva"]).The Cast System: Inside your User.php model, permissions => array tells Laravel to convert that database JSON text string automatically into a real PHP array whenever you load the user.The Check (hasPermission): The custom function uses in_array() to scan your user's permission array. If the required permission string exists, it grants access.Fallback Redirect Strategy: If a user tries to access a page they are not allowed to see, getFirstRedirectRoute() loops through a checklist of routes in order of priority. It safely redirects them to the very first page they actually have permission to view.Step-by-Step Implementation1. Database MigrationTo make this work, you must add a permissions column to your existing users table.Run this command in your terminal:bashphp artisan make:migration add_permissions_to_users_table --table=users
Use code with caution.Open the newly created migration file and add the column:phppublic function up(): void

        // Adds a text or json column that can be empty (nullable)
        $table->json('permissions')->nullable()->after('password');
    });
}

public function updatePermissions(Request $request, $id)
{
    $user = User::findOrFail($id);

    // Save permissions directly as a plain PHP array
    $user->permissions = ['dashboard', 'registrations', 'events'];
    $user->save();

    return response()->json(['message' => 'Permissions updated!']);
}
